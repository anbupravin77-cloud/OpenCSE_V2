// --- ADS.TXT ---
app.get('/ads.txt', (req, res) => {
  const adsTxtPath = path.join(process.cwd(), 'public', 'ads.txt');

  if (fs.existsSync(adsTxtPath)) {
    res.type('text/plain').send(fs.readFileSync(adsTxtPath, 'utf8'));
  } else {
    res.status(404).type('text/plain').send('ads.txt not found');
  }
});
